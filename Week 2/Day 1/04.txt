3. // Snippet 3 — should wait for data before logging
async function run() {
  const data = fetchData(); // bug
  console.log(data.title);
}
async function fetchData() {
  const res = await fetch('https://jsonplaceholder.typicode.com/posts/1');
  return res.json();
}


answer:
-------

- fetchData() is async, so it returns a Promise, not actual data.



async function run() {
  const data = await fetchData(); 
  console.log(data.title);
}

async function fetchData() {
  const res = await fetch('https://jsonplaceholder.typicode.com/posts/1');
