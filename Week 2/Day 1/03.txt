1.Last week you used .then().catch() to fetch data. This week, rewrite it the modern way. Almost all professional code uses async/await.

function loadPost(id) {
  return fetch(`https://jsonplaceholder.typicode.com/posts/${id}`)
    .then(response => {
      if (!response.ok) throw new Error("Post not found");
      return response.json();
    })
    .then(data => {
      console.log(data.title);
      return data;
    })
    .catch(error => {
      console.error("Failed:", error.message);
    });
}


answer :

 async function loadPost(id) {
  try {
    const response = await fetch(`https://jsonplaceholder.typicode.com/posts/${id}`);

    if (!response.ok) {
      throw new Error("Post not found");
    }

    const data = await response.json();

    console.log(data.title);

    return data;

  } catch (error) {
    console.error("Failed:", error.message);
  }
}

2. Write an async function that fetches post #1 AND its comments simultaneously (not one after the other). Both URLs:

https://jsonplaceholder.typicode.com/posts/1
https://jsonplaceholder.typicode.com/posts/1/comments
Return both as an object: { post, comments }


answer:

async function getPostAndComments() {
  try {
    const [postRes, commentsRes] = await Promise.all([
      fetch("https://jsonplaceholder.typicode.com/posts/1"),
      fetch("https://jsonplaceholder.typicode.com/posts/1/comments")
    ]);

    if (!postRes.ok) throw new Error("Post not found");
    if (!commentsRes.ok) throw new Error("Comments not found");

    const post = await postRes.json();
    const comments = await commentsRes.json();

    return { post, comments };

  } catch (error) {
    console.error("Failed:", error.message);
  }
}
