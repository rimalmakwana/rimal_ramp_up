// Supposed to double every number in an array
function doubleAll(numbers) {
  return numbers.map(n => {
    n * 2; // bug here
  });
}


solution : 

- here n * 2 inside the {} so need to return dont write directly

function doubleAll(numbers) {
  return numbers.map(n => {
   return  n * 2;  // put the return keyword here
  });
}

             OR

function doubleAll(numbers) {
  return numbers.map(n => 
   return  n * 2;  // remove a {}
  );
}

  if (num % 2 = 0) { // bug here

}
